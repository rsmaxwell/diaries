+def protected_retention_candidates(entries, base_keep, protected_stamp):
+    """Retain the parent and all newer snapshots, independent of bucket policy.
+
+    The caller must already have verified all snapshot receipts and the parent UUID.
+    This helper has no Btrfs or filesystem side effects and authorises no pruning.
+    """
+    require(isinstance(protected_stamp, str) and STAMP.fullmatch(protected_stamp),
+            'Invalid protected-parent timestamp')
+    names = [name for name, _, _ in entries]
+    require(all(isinstance(name, str) and STAMP.fullmatch(name) for name in names),
+            'Invalid receiver snapshot timestamp')
+    require(len(names) == len(set(names)) and names == sorted(names, reverse=True),
+            'Unsorted or duplicate received snapshot inventory')
+    require(protected_stamp in names, 'Protected parent missing from received inventory')
+    require(set(base_keep).issubset(set(names)), 'Retention keep set references unknown snapshots')
+    keep = set(base_keep)
+    keep.update(name for name in names if name >= protected_stamp)
+    candidates = [name for name in names if name not in keep]
+    require(all(name < protected_stamp for name in candidates),
+            'Receiver candidate is not older than protected common parent')
+    return keep, candidates
