#!/bin/bash
set -euo pipefail

usage() {
    echo "Usage: $0 [--apply|--help]"
    echo "Default: Ansible check mode (--check --diff); --apply performs a real deployment."
}

mode="check"
case "${1:-}" in
    "") ;;
    --apply) mode="apply" ;;
    --help|-h) usage; exit 0 ;;
    *) usage >&2; exit 2 ;;
esac
if (( $# > 1 )); then
    usage >&2
    exit 2
fi

cd "$HOME/playbooks"

args=(
    local-backup.yaml
    --limit mars
    --vault-password-file "$HOME/.vault_pass.txt"
)
if [[ "$mode" == "check" ]]; then
    args+=(--check --diff)
else
    args+=(--diff)
fi

ansible-playbook "${args[@]}"
