IMAGE="$WORK/filesystem.img"

echo '=== Recheck disposable paths ==='

test ! -e "$WORK"

sudo mkdir -- "$WORK"

echo '=== Format disposable Btrfs filesystem ==='

sudo mkfs.btrfs -f -L diaries-qgroup-test "$LOOP"
