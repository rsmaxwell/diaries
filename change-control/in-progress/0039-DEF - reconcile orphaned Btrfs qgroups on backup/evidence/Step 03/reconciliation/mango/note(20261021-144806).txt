richard@mango:~$ set -euo pipefail

MOUNT=/mnt/diaries-qgroup-reproduction
IMAGE=/var/tmp/diaries-qgroup-reproduction/filesystem.img
EXPECTED_UUID=8346c9fd-6dbb-45b6-94dc-e8b4728e2757

echo '=== 1. Verify backing image and loop device ==='

test -f "$IMAGE"

LOOP=$(sudo losetup -j "$IMAGE" | cut -d: -f1)
test -n "$LOOP"

echo "Loop device: $LOOP"
sudo losetup --list "$LOOP"

echo '=== 2. Verify mounted filesystem ==='

test "$(findmnt -rn -o SOURCE --target "$MOUNT")" = "$LOOP"
test "$(findmnt -rn -o FSTYPE --target "$MOUNT")" = btrfs

UUID=$(sudo btrfs filesystem show "$MOUNT" |
  sed -n 's/.*uuid: \([^ ]*\).*/\1/p' | head -n 1)

test "$UUID" = "$EXPECTED_UUID"
echo "Verified filesystem UUID: $UUID"

echo '=== 3. Full subvolume inventory ==='

sudo btrfs subvolume list -a "$MOUNT"

echo '=== 4. Complete qgroup inventory ==='

sudo btrfs qgroup show -pcre "$MOUNT"

echo '=== 5. Confirm both deleted snapshots remain absent ==='

test ! -e "$MOUNT/snapshot-a"
test ! -e "$MOUNT/snapshot-b"

echo 'Both deleted snapshots are absent'

echo '=== 6. Check pending deletions ==='

PENDING=$(sudo btrfs subvolume list -d "$MOUNT")
printf '%s\n' "$PENDING"
test -z "$PENDING"

echo '=== 7. Check quota rescan status ==='

sudo btrfs quota rescan -s "$MOUNT"

echo '=== 8. Record filesystem allocation ==='

sudo btrfs filesystem usage -T "$MOUNT"
