richard@mango:~$ set -euo pipefail

MOUNT=/mnt/diaries-qgroup-reproduction
IMAGE=/var/tmp/diaries-qgroup-reproduction/filesystem.img
EXPECTED_UUID=8346c9fd-6dbb-45b6-94dc-e8b4728e2757

echo '=== Gate C: Verify disposable filesystem identity ==='

test -f "$IMAGE"

LOOP=$(sudo losetup -j "$IMAGE" | cut -d: -f1)
test -n "$LOOP"

test "$(sudo losetup -n -O BACK-FILE "$LOOP")" = "$IMAGE"
test "$(findmnt -rn -o SOURCE --target "$MOUNT")" = "$LOOP"
test "$(findmnt -rn -o FSTYPE --target "$MOUNT")" = btrfs

UUID=$(sudo btrfs filesystem show "$MOUNT" |
  sed -n 's/.*uuid: \([^ ]*\).*/\1/p' | head -n 1)

test "$UUID" = "$EXPECTED_UUID"

echo '=== Verify exact subvolume inventory ==='

SUBVOLUMES=$(sudo btrfs subvolume list -a "$MOUNT")
printf '%s\n' "$SUBVOLUMES"

echo "$SUBVOLUMES" | grep -Eq '^ID 256 .* path source-a$'
echo "$SUBVOLUMES" | grep -Eq '^ID 258 .* path source-b$'

test "$(printf '%s\n' "$SUBVOLUMES" | grep -c '^ID ')" -eq 2

echo '=== Verify qgroup baseline ==='

BEFORE=$(sudo btrfs qgroup show -pcre "$MOUNT")
printf '%s\n' "$BEFORE"

EXPECTED_ACTIVE=$(
  printf '%s\n' "$BEFORE" |
    grep -E '^0/(5|256|258)[[:space:]]'
)

test "$(printf '%s\n' "$EXPECTED_ACTIVE" | wc -l)" -eq 3

printf '%s\n' "$BEFORE" |
  grep -Eq '^0/259[[:space:]]+0\.00B[[:space:]]+0\.00B[[:space:]]+1\.00TiB[[:space:]]+none[[:space:]]+-[[:space:]]+-[[:space:]]+<stale>$'

if printf '%s\n' "$BEFORE" | grep -Eq '^0/257[[:space:]]'; then
    echo 'ABORT: Previously removed qgroup 0/257 has reappeared'
    exit 1
fi

echo '=== Verify no pending deletion or quota rescan ==='

test ! -e "$MOUNT/snapshot-a"
test ! -e "$MOUNT/snapshot-b"
test -z "$(sudo btrfs subvolume list -d "$MOUNT")"

RESCAN=$(sudo btrfs quota rescan -s "$MOUNT")
test "$RESCAN" = 'no rescan operation in progress'

echo '=== Destroy ONLY stale qgroup 0/259 ==='

sudo btrfs qgroup destroy 0/259 "$MOUNT"

echo '=== Compare post-destruction state ==='

AFTER=$(sudo btrfs qgroup show -pcre "$MOUNT")
printf '%s\n' "$AFTER"

if printf '%s\n' "$AFTER" |
    grep -Eq '^0/(257|259)[[:space:]]'; then
    echo 'FAIL: A stale qgroup remains'
    exit 1
fi

ACTUAL_ACTIVE=$(
  printf '%s\n' "$AFTER" |
    grep -E '^0/(5|256|258)[[:space:]]'
)

test "$ACTUAL_ACTIVE" = "$EXPECTED_ACTIVE"

test "$(printf '%s\n' "$AFTER" | grep -c '^0/')" -eq 3

echo '=== Verify pending deletions and quota status ==='

test -z "$(sudo btrfs subvolume list -d "$MOUNT")"

RESCAN=$(sudo btrfs quota rescan -s "$MOUNT")
test "$RESCAN" = 'no rescan operation in progress'

echo '=== GATE C COMPLETE ==='
