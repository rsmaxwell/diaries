richard@mango:~/playbooks$ set -euo pipefail

MOUNT=/mnt/diaries-qgroup-reproduction

echo '=== Disposable Btrfs filesystem ==='
sudo btrfs filesystem show "$MOUNT"

echo '=== Experimental qgroups ==='
sudo btrfs qgroup show -pcre "$MOUNT" |
  grep -E '^Qgroupid|^--------|^0/(257|259)[[:space:]]'

echo '=== Confirm snapshots are absent ==='
for NAME in snapshot-a snapshot-b; do
    if [ -e "$MOUNT/$NAME" ]; then
        echo "UNEXPECTED: $NAME still exists"
        exit 1
    fi
    echo "ABSENT: $NAME"
done

echo '=== Pending deletions ==='
sudo btrfs subvolume list -d "$MOUNT"

echo '=== Quota rescan status ==='
sudo btrfs quota rescan -s "$MOUNT"
