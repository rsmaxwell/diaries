  head -n 40

sudo snapper -c root list --disable-used-space |
  head -n 25

sudo snapper -c @data list --disable-used-space |
  head -n 25

echo '=== Historical snapshot directories ==='

for DIR in /.snapshots /@data/.snapshots; do
    echo "=== $DIR ==="

    if sudo test -d "$DIR"; then
        sudo find "$DIR" -mindepth 1 -maxdepth 1 -type d \
          -printf '%f\n' | sort -n | tail -n 20
    else
        echo 'NOT PRESENT'
    fi
done
